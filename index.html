// --- TWO-WAY SYNC RATING FUNCTIONS --- async function saveRating(recipeKey, score, e) { if (e) e.stopPropagation(); // Find recipe object const recipe = RECIPES.find(r => r.title === recipeKey || r.id === recipeKey); const title = recipe ? recipe.title : recipeKey; // Toggle off if clicking the current rating const current = ratingsMap[title] || 0; const newRating = (current === score) ? 0 : score; // 1. Instant local update on screen if (newRating === 0) { delete ratingsMap[title]; if (recipe) recipe.rating = 0; } else { ratingsMap[title] = newRating; if (recipe) recipe.rating = newRating; } // Persist locally for instant offline/reloads localStorage.setItem(RATINGS_KEY, JSON.stringify(ratingsMap)); render(); if (activeRecipe && (activeRecipe.title === title || activeRecipe.id === recipeKey)) { renderModalRating(title); } // 2. Push update directly to Google Sheet try { await fetch(APPS_SCRIPT_URL, { method: "POST", mode: "no-cors", // Bypass browser CORS restrictions for Apps Script headers: { "Content-Type": "application/json" }, body: JSON.stringify({ title: title, rating: newRating }) }); } catch (err) { console.warn("Failed to sync rating to Google Sheet:", err); } } // When recipes load from Google Sheets, hydrate the ratings function hydrateRatingsFromSheet(recipes) { recipes.forEach(r => { if (r.rating && r.rating > 0) { ratingsMap[r.title] = parseInt(r.rating); } }); localStorage.setItem(RATINGS_KEY, JSON.stringify(ratingsMap)); }
